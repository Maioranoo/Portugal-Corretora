```ts
import { readFileSync } from 'node:fs';
import { describe, expect, test } from 'vitest';

const txt = readFileSync('.vercel/output/static/robots.txt', 'utf-8');

// Grupos do robots.txt (RFC 9309): linhas User-agent seguidas compartilham as regras abaixo delas.
function groups(source: string) {
  const out: { agents: string[]; disallow: string[] }[] = [];
  let current: { agents: string[]; disallow: string[] } | null = null;
  let lastWasAgent = false;
  for (const raw of source.split('\n')) {
    const line = raw.replace(/#.*/, '').trim();
    if (!line) continue;
    const [key, ...rest] = line.split(':');
    const value = rest.join(':').trim();
    if (/^user-agent$/i.test(key)) {
      if (!lastWasAgent || !current) out.push((current = { agents: [], disallow: [] }));
      current.agents.push(value.toLowerCase());
      lastWasAgent = true;
    } else {
      if (/^disallow$/i.test(key) && current) current.disallow.push(value);
      lastWasAgent = false;
    }
  }
  return out;
}
const rulesFor = (agent: string) =>
  groups(txt).find((g) => g.agents.includes(agent.toLowerCase())) ?? groups(txt).find((g) => g.agents.includes('*'))!;

describe('robots.txt', () => {
  test.each(['GPTBot', 'ClaudeBot', 'Google-Extended', 'CCBot', 'Applebot-Extended', 'Meta-ExternalAgent', 'Bytespider'])(
    'robô de treinamento %s é bloqueado',
    (agent) => {
      expect(rulesFor(agent).disallow).toContain('/');
    },
  );
  test.each(['Googlebot', 'Bingbot', 'OAI-SearchBot', 'ChatGPT-User', 'Claude-SearchBot', 'PerplexityBot'])(
    'robô de busca %s continua liberado',
    (agent) => {
      expect(rulesFor(agent).disallow).not.toContain('/');
    },
  );
  test('aponta o sitemap', () => {
    expect(txt).toContain('Sitemap: https://www.portugalcorretora.com.br/sitemap.xml');
  });
});
```

```ts
import { PRODUCTS, type Faq } from '../content/products';
import { AREA_SERVED, COMPANY } from '../content/site';

export const SITE_URL = 'https://www.portugalcorretora.com.br';
export const ORG_ID = `${SITE_URL}/#corretora`;
export const WEBSITE_ID = `${SITE_URL}/#site`;

export type JsonLdNode = Record<string, unknown>;
export interface Crumb {
  name: string;
  path: string;
}

const abs = (path: string) => new URL(path, SITE_URL).href;
const phone = `+${COMPANY.whatsapp}`;
const cities = () =>
  AREA_SERVED.map((name) => ({ '@type': 'City', name, containedInPlace: { '@type': 'State', name: 'São Paulo' } }));

// Só dados reais: nada de nota, prêmio ou certificação que não esteja publicado e comprovado.
export function organizationNode(): JsonLdNode {
  return {
    '@type': 'InsuranceAgency',
    '@id': ORG_ID,
    name: COMPANY.name,
    legalName: COMPANY.legalName,
    taxID: COMPANY.cnpj,
    identifier: { '@type': 'PropertyValue', propertyID: 'SUSEP', value: COMPANY.susep },
    url: abs('/'),
    logo: { '@type': 'ImageObject', url: abs('/icon-512.png'), width: 512, height: 512 },
    image: abs('/og.v2.jpg'),
    telephone: phone,
    email: COMPANY.email,
    address: {
      '@type': 'PostalAddress',
      streetAddress: `${COMPANY.address.street}, ${COMPANY.address.district}`,
      addressLocality: COMPANY.address.city,
      addressRegion: COMPANY.address.state,
      postalCode: COMPANY.address.zip,
      addressCountry: 'BR',
    },
    hasMap: COMPANY.address.mapsUrl,
    areaServed: cities(),
    openingHoursSpecification: [
      {
        '@type': 'OpeningHoursSpecification',
        dayOfWeek: ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday'],
        opens: '09:00',
        closes: '18:00',
      },
    ],
    contactPoint: {
      '@type': 'ContactPoint',
      contactType: 'customer service',
      telephone: phone,
      email: COMPANY.email,
      availableLanguage: 'Portuguese',
    },
    knowsAbout: PRODUCTS.map((p) => p.name),
    sameAs: [COMPANY.social.facebook, COMPANY.social.instagram],
  };
}

export function websiteNode(): JsonLdNode {
  return {
    '@type': 'WebSite',
    '@id': WEBSITE_ID,
    url: abs('/'),
    name: COMPANY.name,
    inLanguage: 'pt-BR',
    publisher: { '@id': ORG_ID },
  };
}

export function breadcrumbNode(items: readonly Crumb[], path: string): JsonLdNode {
  return {
    '@type': 'BreadcrumbList',
    '@id': `${abs(path)}#trilha`,
    itemListElement: items.map((c, i) => ({ '@type': 'ListItem', position: i + 1, name: c.name, item: abs(c.path) })),
  };
}

export function serviceNode(input: { name: string; path: string; description: string }): JsonLdNode {
  return {
    '@type': 'Service',
    '@id': `${abs(input.path)}#servico`,
    name: input.name,
    serviceType: input.name,
    description: input.description,
    url: abs(input.path),
    provider: { '@id': ORG_ID },
    areaServed: cities(),
  };
}

export function faqNode(faqs: readonly Faq[], path: string): JsonLdNode {
  return {
    '@type': 'FAQPage',
    '@id': `${abs(path)}#faq`,
    mainEntity: faqs.map((f) => ({ '@type': 'Question', name: f.q, acceptedAnswer: { '@type': 'Answer', text: f.a } })),
  };
}

export function buildGraph(nodes: readonly JsonLdNode[]) {
  return { '@context': 'https://schema.org', '@graph': [organizationNode(), websiteNode(), ...nodes] };
}

export function serializeJsonLd(data: unknown): string {
  return JSON.stringify(data).replace(/</g, '\\u003c');
}
```

```ts
const HOST = /^[a-z0-9.-]{1,100}$/;
const CONTROL = /[\u0000-\u001F\u007F]/;

export function safeSessionStorage(): Storage | null {
  try {
    return window.sessionStorage;
  } catch {
    return null;
  }
}

export function referrerHost(referrer: string, ownHost: string): string {
  if (!referrer) return '';
  try {
    const host = new URL(referrer).hostname.toLowerCase();
    return host === ownHost.toLowerCase() ? INTERNAL : host;
  } catch {
    return '';
  }
}

export function sanitizeArrival(raw: unknown): Arrival {
  const r = raw && typeof raw === 'object' ? (raw as Record<string, unknown>) : {};
  const referrer = typeof r.referrer === 'string' && HOST.test(r.referrer) ? r.referrer : '';
  const l = r.landing;
  const landing =
    typeof l === 'string' && l.startsWith('/') && !l.startsWith('//') && l.length <= 200 && !CONTROL.test(l) ? l : '/';
  return { referrer, landing };
}

// A primeira página da sessão vale: ao navegar dentro do site, o navegador passa a informar o próprio site como origem.
export function captureArrival(
  storage: Pick<Storage, 'getItem' | 'setItem'> | null,
  referrer: string,
  ownHost: string,
  pathname: string,
): Arrival {
  try {
    const saved = storage?.getItem(ORIGIN_STORAGE_KEY);
    if (saved) return sanitizeArrival(JSON.parse(saved));
  } catch {
    /* armazenamento indisponível ou corrompido: calcula de novo */
  }
  const arrival = sanitizeArrival({ referrer: referrerHost(referrer, ownHost), landing: pathname });
  try {
    storage?.setItem(ORIGIN_STORAGE_KEY, JSON.stringify(arrival));
  } catch {
    /* segue sem guardar */
  }
  return arrival;
}

const RULES: readonly [RegExp, string][] = [
  [/(^|\.)chatgpt\.com$|(^|\.)openai\.com$/, 'ChatGPT'],
  [/(^|\.)perplexity\.ai$/, 'Perplexity'],
  [/^gemini\.google\.com$/, 'Gemini'],
  [/(^|\.)copilot\.microsoft\.com$/, 'Copilot'],
  [/(^|\.)google\.[a-z.]+$|googlequicksearchbox/, 'Google (busca)'],
  [/(^|\.)bing\.com$/, 'Bing (busca)'],
  [/(^|\.)duckduckgo\.com$/, 'DuckDuckGo (busca)'],
  [/(^|\.)yahoo\.com$/, 'Yahoo (busca)'],
  [/(^|\.)instagram\.com$/, 'Instagram'],
  [/(^|\.)facebook\.com$/, 'Facebook'],
];
const match = (value: string) => RULES.find(([re]) => re.test(value))?.[1];

export function describeOrigin(arrival: Arrival, utm: Utm): string {
  const source = utm.utm_source?.toLowerCase();
  if (source) {
    return match(source) ?? `Campanha (${[utm.utm_source, utm.utm_campaign].filter(Boolean).join(' / ')})`;
  }
  if (!arrival.referrer) return 'Acesso direto ou link sem origem (WhatsApp, apps)';
  if (arrival.referrer === INTERNAL) return 'Não identificada (navegação interna)';
  return match(arrival.referrer) ?? `Outro site (${arrival.referrer})`;
}
```
